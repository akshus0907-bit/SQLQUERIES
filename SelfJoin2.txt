mysql> select m.employee_name as manager ,
    -> avg(e.salary) as Avg_sal ,
    -> count(e.employee_id) as Total_emp
    -> from Employee m join Employee e on e.manager_id=m.employee_id group by m.employee_name having count(e.employee_id)>=2 and avg(e.salary)>50000;
