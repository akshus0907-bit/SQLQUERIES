mysql> SELECT
    ->     e1.employee_name AS Employee1,
    ->     e2.employee_name AS Employee2,
    ->     m.employee_name AS Manager
    -> FROM Employee e1
    -> JOIN Employee e2
    -> ON e1.manager_id = e2.manager_id
    -> JOIN Employee m
    -> ON e1.manager_id = m.employee_id
    -> WHERE e1.employee_id < e2.employee_id;
